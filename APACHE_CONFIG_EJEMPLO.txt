# Agregar AL FINAL de C:/xampp/apache/conf/httpd.conf, con permiso del administrador.
# Hace que http://192.168.20.132/clinica_avisos/ sirva SOLO el directorio public.
# Importante: no crear un Alias que apunte a la raiz del proyecto.

<Directory "C:/xampp/htdocs/clinica_avisos">
    AllowOverride None
    Require all denied
    Options -Indexes
</Directory>

Alias "/clinica_avisos" "C:/xampp/htdocs/clinica_avisos/public"

<Directory "C:/xampp/htdocs/clinica_avisos/public">
    AllowOverride None
    Require all granted
    Options -Indexes
    DirectoryIndex index.php
</Directory>
